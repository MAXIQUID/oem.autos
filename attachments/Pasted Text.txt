<script>
class Cache {
  constructor(){this.prefix='oem_cache_v2_';this.expiry=3600*1000;}
  get(k){try{const v=JSON.parse(localStorage.getItem(this.prefix+k));if(!v)return null;if(Date.now()-v.timestamp>this.expiry){localStorage.removeItem(this.prefix+k);return null;}return v.value;}catch{return null;}}
  set(k,d){try{localStorage.setItem(this.prefix+k,JSON.stringify({value:d,timestamp:Date.now()}));}catch{}}
  saveState(state){localStorage.setItem('oem_state',JSON.stringify({vin:state.vin,year:state.year,make:state.make,model:state.model,option:state.option}));}
  loadState(){try{return JSON.parse(localStorage.getItem('oem_state'))||{};}catch{return {};}}
}
const cache=new Cache();
const els={vin:document.getElementById('vin-input'),btn:document.getElementById('btn-decode'),year:document.getElementById('year-carousel'),make:document.getElementById('make-carousel'),model:document.getElementById('model-carousel'),option:document.getElementById('option-carousel'),result:document.getElementById('result-content'),sticky:document.getElementById('sticky-header'),sentinel:document.getElementById('sticky-sentinel')};
const state={vin:'',year:null,make:null,model:null,option:null,years:[],makes:[],models:[],options:[],details:null,loading:false};
new IntersectionObserver(e=>{els.sticky.classList.toggle('is-stuck',!e[0].isIntersecting);},{threshold:[1]}).observe(els.sentinel);
async function fetchJSON(url){const cached=cache.get(url);if(cached)return cached;try{const r=await fetch(url);const j=await r.json();cache.set(url,j);return j;}catch{return null;}}
async function fetchXML(url){const cached=cache.get(url);if(cached)return cached;try{const r=await fetch(url);const txt=await r.text();const doc=new DOMParser().parseFromString(txt,'application/xml');cache.set(url,doc);return doc;}catch{return null;}}
function saveState(){cache.saveState(state);}
function setPills(container,items,value,onClick){container.innerHTML='';items.forEach(i=>{const p=document.createElement('div');p.className='pill'+(i.value===value?' active':'');p.textContent=i.text;p.onclick=()=>onClick(i.value);container.appendChild(p);});if(container.querySelector('.active'))container.querySelector('.active').scrollIntoView({inline:'center',behavior:'smooth'});}
async function loadYears(){state.years=[];for(let y=2026;y>=1984;y--)state.years.push({text:y,value:y});setPills(els.year,state.years,state.year,val=>{state.year=val;state.make=null;state.model=null;state.option=null;saveState();loadMakes();});}
async function loadMakes(){if(!state.year)return;state.makes=[];const doc=await fetchXML(`https://www.fueleconomy.gov/ws/rest/vehicle/menu/make?year=${state.year}`);if(doc){state.makes=Array.from(doc.getElementsByTagName('menuItem')).map(x=>({text:x.querySelector('text').textContent,value:x.querySelector('value').textContent}));}setPills(els.make,state.makes,state.make,val=>{state.make=val;state.model=null;state.option=null;saveState();loadModels();});}
async function loadModels(){if(!state.year||!state.make)return;state.models=[];const doc=await fetchXML(`https://www.fueleconomy.gov/ws/rest/vehicle/menu/model?year=${state.year}&make=${state.make}`);if(doc){state.models=Array.from(doc.getElementsByTagName('menuItem')).map(x=>({text:x.querySelector('text').textContent,value:x.querySelector('value').textContent}));}setPills(els.model,state.models,state.model,val=>{state.model=val;state.option=null;saveState();loadOptions();});}
async function loadOptions(){if(!state.year||!state.make||!state.model)return;state.options=[];const doc=await fetchXML(`https://www.fueleconomy.gov/ws/rest/vehicle/menu/options?year=${state.year}&make=${state.make}&model=${state.model}`);if(doc){state.options=Array.from(doc.getElementsByTagName('menuItem')).map(x=>({text:x.querySelector('text').textContent,value:x.querySelector('value').textContent}));}setPills(els.option,state.options,state.option,val=>{state.option=val;saveState();loadDetails();});}
async function loadDetails(){if(!state.option)return;els.result.innerHTML='<p style="text-align:center;">Loading...</p>';const doc=await fetchXML(`https://www.fueleconomy.gov/ws/rest/vehicle/${state.option}`);if(doc){const data={};Array.from(doc.documentElement.children).forEach(c=>data[c.tagName]=c.textContent);state.details=data;renderDetails();}}
function renderDetails(){if(!state.details){els.result.innerHTML='<p>No data</p>';return;}const d=state.details;const vin=state.vin.toUpperCase();els.result.innerHTML=`<div class="vin-viz"><div class="vin-chunk c-wmi"><span>${vin.slice(0,3)}</span><span class="lbl">WMI</span></div><div class="vin-chunk c-vds"><span>${vin.slice(3,9)}</span><span class="lbl">VDS</span></div><div class="vin-chunk c-vis"><span>${vin.slice(9,17)}</span><span class="lbl">VIS</span></div></div><h2>${d.ModelYear||state.year||''} ${d.Make||state.make||''} ${d.Model||state.model||''}</h2><div class="data-grid"><div class="data-item"><small>Manufacturer</small><strong>${d.Manufacturer||''}</strong></div><div class="data-item"><small>Engine</small><strong>${d.EngineCylinders||''} Cyl ${d.DisplacementL||''}L</strong></div><div class="data-item"><small>Fuel</small><strong>${d.FuelTypePrimary||''}</strong></div><div class="data-item"><small>Body</small><strong>${d.BodyClass||''}</strong></div><div class="data-item"><small>Drive</small><strong>${d.DriveType||''}</strong></div></div>`;}
async function decodeVIN(){const vin=els.vin.value.trim();if(vin.length!==17)return alert('VIN must be 17 characters');state.vin=vin;saveState();els.result.innerHTML='<p style="text-align:center;">Decoding VIN...</p>';const data=await fetchJSON(`https://vpic.nhtsa.dot.gov/api/vehicles/decodevin/${vin}?format=json`);if(data&&data.Results){const clean={};data.Results.forEach(r=>{if(r.Value&&r.Value!=='null')clean[r.Variable.replace(/ /g,'')]=r.Value;});state.details=clean;renderDetails();}}
els.btn.addEventListener('click',decodeVIN);
els.vin.addEventListener('keypress',e=>{if(e.key==='Enter')decodeVIN();});
(async()=>{loadYears();const saved=cache.loadState();if(saved.vin){state.vin=saved.vin;els.vin.value=state.vin;decodeVIN();}if(saved.year){state.year=saved.year;loadMakes();}if(saved.make){state.make=saved.make;loadModels();}if(saved.model){state.model=saved.model;loadOptions();}if(saved.option){state.option=saved.option;loadDetails();}})();
</script>
